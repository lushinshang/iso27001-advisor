# scripts/build_embeddings.py（新建）
"""預先計算 ISO 27001 知識庫的 sentence embedding，存為 numpy 陣列。"""
import json
import numpy as np
from pathlib import Path
from sentence_transformers import SentenceTransformer

BASE = Path(__file__).resolve().parents[1]
DATA = BASE / "data"

def build_text(item):
    """將條文節點的欄位拼接成適合 embed 的字串。"""
    parts = [item.get("subsection", ""), item.get("title", ""), item.get("content", "")]
    return " ".join(p for p in parts if p)

model = SentenceTransformer("paraphrase-multilingual-MiniLM-L12-v2")

with open(DATA / "iso27001_structure.json", encoding="utf-8") as f:
    items = json.load(f)

ids = [item["id"] for item in items]
texts = [build_text(item) for item in items]

print(f"Encoding {len(texts)} items...")
embeddings = model.encode(texts, normalize_embeddings=True, show_progress_bar=True)

np.save(DATA / "embeddings.npy", embeddings)
with open(DATA / "embedding_ids.json", "w", encoding="utf-8") as f:
    json.dump(ids, f, ensure_ascii=False)

print(f"✅ 儲存：data/embeddings.npy ({embeddings.shape}) + data/embedding_ids.json")

    import numpy as np

    _EMBEDDING_AVAILABLE = False

# 在 load_data() 之後，search() 之前，新增下列三個方法
    def _load_embeddings(self):
        """延遲載入 embedding 索引（首次呼叫 hybrid_search 時才載入）。"""
        if not _EMBEDDING_AVAILABLE:
            raise ImportError("需要 sentence-transformers：pip install sentence-transformers")
        if self._embeddings is not None:
            return
        base = Path(self.json_path).parent
        emb_path = base / "embeddings.npy"
        ids_path = base / "embedding_ids.json"
        if not emb_path.exists():
            raise FileNotFoundError(
                f"找不到 {emb_path}，請先執行：python3 scripts/build_embeddings.py"
            )
        self._embeddings = np.load(str(emb_path))
        with open(ids_path, encoding="utf-8") as f:
            self._embed_ids = json.load(f)
        self._embed_model = SentenceTransformer("paraphrase-multilingual-MiniLM-L12-v2")

    def embedding_search(self, query, limit=10):
        """語意向量搜尋，回傳 [{'item': dict, 'score': float}] 按相似度排序。"""
        self._load_embeddings()
        q_vec = self._embed_model.encode([query], normalize_embeddings=True)[0]
        scores = self._embeddings @ q_vec          # cosine similarity（已 normalize）
        top_idxs = scores.argsort()[::-1][:limit]
        results = []
        for idx in top_idxs:
            item_id = self._embed_ids[idx]
            item = self._id_index.get(item_id)
            if item:
                results.append({"item": item, "score": float(scores[idx])})
        return results

    def hybrid_search(self, query, limit=4, keyword_weight=0.5, embed_weight=0.5):
        """Hybrid Search：RRF 融合關鍵字搜尋 + 語意搜尋。

        RRF 公式：score(d) = Σ 1 / (k + rank(d))，k=60（業界慣例）
        兩路各取 top 20 候選，融合後取 top limit。
        """
        POOL = 20
        K = 60

        kw_results  = self.search(query, limit=POOL)
        emb_results = self.embedding_search(query, limit=POOL)

        kw_rank  = {r["item"]["id"]: i for i, r in enumerate(kw_results)}
        emb_rank = {r["item"]["id"]: i for i, r in enumerate(emb_results)}

        candidate_ids = set(kw_rank) | set(emb_rank)
        rrf_scores = {}
        for cid in candidate_ids:
            kw_s  = keyword_weight / (K + kw_rank[cid])  if cid in kw_rank  else 0.0
            emb_s = embed_weight   / (K + emb_rank[cid]) if cid in emb_rank else 0.0
            rrf_scores[cid] = kw_s + emb_s

        ranked = sorted(candidate_ids, key=lambda x: rrf_scores[x], reverse=True)[:limit]
        return [{"item": self._id_index[cid], "score": rrf_scores[cid]}
                for cid in ranked if cid in self._id_index]

Step 3-2：建立 scripts/build_knowledge_graph.py（LLM 輔助生成候選邊）

用 Ollama 批次分析每個控制項，自動建議相關聯的控制項，輸出候選邊供人工審核。

# scripts/build_knowledge_graph.py（新建）
"""用 LLM 批次分析 ISO 27001 條文，生成關聯邊候選，供人工審核後加入 knowledge_graph.json。"""
import json, sys
from pathlib import Path

BASE = Path(__file__).resolve().parents[1]
sys.path.insert(0, str(BASE / "src"))

from iso27001_advisor.llm import call_ollama, load_dotenv

load_dotenv()

with open(BASE / "data" / "iso27001_structure.json", encoding="utf-8") as f:
    items = json.load(f)

idx = {i["id"]: i for i in items}

PROMPT_TMPL = """以下是 ISO 27001:2022 的一個控制項：

ID: {id}
名稱: {subsection}
內容: {content}

請列出 3-5 個與此控制項「最密切相關」的其他控制項 ID（格式：control_X.X 或 clause_X.X）。
「密切相關」的定義：稽核員在查核此控制項時，通常需要同時確認的其他控制項。

只輸出 JSON 陣列，不要其他說明：["control_X.X", "clause_X.X"]"""

candidates = {}
# 只處理有小數點的真實條文（排除父層節點）
targets = [i for i in items if "." in i["id"] and len(i["content"]) > 10]

for t in targets[:30]:   # 先跑前 30 個驗證品質，確認 OK 後移除 [:30]
    prompt = PROMPT_TMPL.format(**t)
    try:
        raw = call_ollama(prompt, "你是 ISO 27001 專家，只輸出 JSON 陣列。")
        # 嘗試解析 JSON
        import re
        m = re.search(r'\[.*?\]', raw, re.DOTALL)
        if m:
            related = json.loads(m.group())
            # 過濾：只保留知識庫中存在的 ID
            valid = [r for r in related if r in idx]
            if valid:
                candidates[t["id"]] = valid
                print(f"  {t['id']}: {valid}")
    except Exception as e:
        print(f"  ⚠️  {t['id']} 失敗: {e}", file=sys.stderr)

out = BASE / "data" / "knowledge_graph_candidates.json"
with open(out, "w", encoding="utf-8") as f:
    json.dump(candidates, f, ensure_ascii=False, indent=2)

    def _load_kg(self):
        """載入 knowledge_graph.json（若不存在則靜默跳過）。"""
        kg_path = Path(self.json_path).parent / "knowledge_graph.json"
        if kg_path.exists():
            with open(kg_path, encoding="utf-8") as f:
                data = json.load(f)
            self._kg = data.get("edges", {})

    def graph_augmented_search(self, query, limit=4):
        """Graph-Augmented 搜尋：先做 hybrid_search，再做 1-hop KG 擴展 + rerank。

        流程：
        1. hybrid_search 取 top-(limit) 初始結果
        2. 對每個結果，查 KG 取出相關節點（1-hop）
        3. 全部候選（初始 + 1-hop）用 hybrid 分數排序
        4. 去重後取 top limit
        """
        try:
            initial = self.hybrid_search(query, limit=limit)
        except (ImportError, FileNotFoundError):
            initial = self.search(query, limit=limit)

        if not self._kg:
            return initial  # KG 未載入，降級到 hybrid

        # 取得初始結果的 ID 與分數
        seen = {r["item"]["id"]: r["score"] for r in initial}

        # 1-hop 擴展：取得所有相鄰節點 ID
        expanded_ids = set()
        for r in initial:
            cid = r["item"]["id"]
            for neighbor in self._kg.get(cid, []):
                if neighbor not in seen:
                    expanded_ids.add(neighbor)

        # 對擴展節點用 embedding 相似度評分（若可用）
        if expanded_ids and self._embeddings is not None:
            try:
                q_vec = self._embed_model.encode([query], normalize_embeddings=True)[0]
                for eid in expanded_ids:
                    if eid in self._id_index:
                        idx_pos = self._embed_ids.index(eid) if eid in self._embed_ids else -1
                        if idx_pos >= 0:
                            sim = float(self._embeddings[idx_pos] @ q_vec)
                            seen[eid] = sim * 0.8   # 1-hop 擴展節點給予 0.8 折扣分
            except Exception:
                pass
        else:
            for eid in expanded_ids:
                if eid in self._id_index:
                    seen[eid] = 0.1  # 無 embedding 時給固定低分

        # 排序 + 去重 + 取 top limit
        ranked = sorted(seen.items(), key=lambda x: x[1], reverse=True)[:limit]
        return [{"item": self._id_index[cid], "score": score}
                for cid, score in ranked if cid in self._id_index]
